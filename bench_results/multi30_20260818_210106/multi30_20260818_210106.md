def is_balanced(s: str) -> bool:
    """
    判断字符串 s 中的三种括号 () [] {} 是否匹配。
    """
    stack = []
    # 定义闭合括号到对应开闭括号的映射
    mapping = {
        ')': '(',
        ']': '[',
        '}': '{'
    }

    for char in s:
        # 1. 如果是开括号，推入栈中
        if char in mapping.values():
            stack.append(char)
        # 2. 如果是闭括号
        elif char in mapping.keys():
            # 检查栈是否为空 (即遇到闭括号但没有对应的开括号)
            if not stack:
                return False
            
            # 弹出栈顶元素，并检查它是否与当前闭括号匹配
            top_element = stack.pop()
            if mapping[char] != top_element:
                return False
        # 3. 如果是其他字符（忽略），则不做任何操作

    # 遍历结束后，如果栈为空，说明所有开括号都找到了对应的闭括号
    return not stack

    """
    Finds the longest common prefix string amongst an array of strings.
    """
    if not strs:
        return ""

    # Use the first string as the reference for comparison
    first_str = strs[0]
    
    # Iterate through the characters (index i) of the first string
    for i in range(len(first_str)):
        char = first_str[i]
        
        # Compare this character against all other strings (starting from index 1)

def is_balanced(s):

def longest_common_prefix(strs):
    if not strs:
        return ""
    shortest = min(strs, key=len)
    for i, char in enumerate(shortest):
        for string in strs:
            if i >= len(string) or string[i] != char:
                return shortest[:i]
    return shortest

def longest_common_prefix(strs):
    if not strs:
        return ""
    prefix = strs[0]
    for s in strs[1:]:

def is_balanced(s: str) -> bool:

    stack = []
